function animateRoute(routePoints) {

  let currentStep = 0;

  function animateSegment() {
    if (currentStep < routePoints.length) {
      polyline.addLatLng(routePoints[currentStep]);
      currentStep++;
      setTimeout(animateSegment, 20); // Adjust speed for smoothness
    } else {
      console.log("Route Animation Complete");
    }
  }
  animateSegment();
}
